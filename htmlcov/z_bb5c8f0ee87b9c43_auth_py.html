29@router.post(

46def issue_token( 

47 credentials: TokenRequest, 

48 session: Annotated[Session, Depends(get_session)],

50 user = session.exec( 

51 select(User).where(func.lower(User.email) == credentials.email.lower()) 

52 ).first() 

53 

54 if user is None: 

55 verify_dummy_password(credentials.password) # однаковий час відповіді 

56 raise HTTPException(status.HTTP_401_UNAUTHORIZED, detail=_INVALID_CREDENTIALS) 

57 if not verify_password(credentials.password, user.password_hash): 

58 # Одне повідомлення для невірного email і пароля: не підказуємо, що саме не так. 

59 raise HTTPException(status.HTTP_401_UNAUTHORIZED, detail=_INVALID_CREDENTIALS) 

60 if user.is_blocked: 

61 # Лише після перевірки пароля: інакше 403 видавав би, що такий email існує. 

62 raise HTTPException(status.HTTP_403_FORBIDDEN, detail="Обліковий запис заблоковано") 

63 

64 return TokenOut( 

65 access_token=create_access_token(str(user.id), user.role.value), 

66 expires_in=int(access_token_ttl().total_seconds()), 

67 user=to_user_out(user), 

68 )
