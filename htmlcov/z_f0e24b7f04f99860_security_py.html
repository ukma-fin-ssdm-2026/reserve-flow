1"""Паролі й JWT за ADR-003 (L5-12, #205). 

2 

3- Паролі зберігаються лише як bcrypt-хеші (pwdlib), SEC-001. 

4- Access-токен — JWT HS256, підписаний секретом зі змінної JWT_SECRET (не в git). 

5 У payload: user_id (sub) і role; час життя — ACCESS_TOKEN_EXPIRE_MINUTES (за ADR 15–30 хв). 

6 

7Модуль не залежить від FastAPI і БД: його використовують deps.py, routes/auth.py, 

8seed.py, mcp_server.py і тести. 

9""" 

10 

11import os 

12from datetime import datetime, timedelta, timezone 

13from functools import lru_cache 

14 

15import jwt 

16from pwdlib import PasswordHash 

17from pwdlib.exceptions import UnknownHashError 

18from pwdlib.hashers.bcrypt import BcryptHasher 

19 

20JWT_ALGORITHM = "HS256" 

21JWT_ISSUER = "reserve-flow" 

22# HS256: ключ має бути не коротшим за 256 біт (RFC 7518, розд. 3.2) 

23MIN_SECRET_LENGTH = 32 

24 

25# bcrypt обробляє не більше 72 байтів пароля; довший bcrypt 5 відхиляє з ValueError 

26BCRYPT_MAX_BYTES = 72 

27 

28_password_hash = PasswordHash((BcryptHasher(),)) 

29 

30 

31class InvalidToken(Exception): 

32 """Підпис, строк дії або вміст токена некоректні (-> 401).""" 

33 

34 

35def access_token_ttl() -> timedelta: 

36 return timedelta(minutes=int(os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "30"))) 

37 

38 

39def _secret() -> str: 

40 """Секрет читається під час виклику, а не імпорту: застосунок і експорт OpenAPI 

41 стартують і без нього, але видати чи перевірити токен без секрету не можна.""" 

42 secret = os.getenv("JWT_SECRET", "") 

43 if len(secret) < MIN_SECRET_LENGTH: 

44 raise RuntimeError( 

45 f"JWT_SECRET не задано або він коротший за {MIN_SECRET_LENGTH} символи. " 

46 'Згенеруйте: python -c "import secrets; print(secrets.token_urlsafe(48))"' 

47 ) 

48 return secret

51def hash_password(password: str) -> str: 

52 if len(password.encode()) > BCRYPT_MAX_BYTES: 

53 raise ValueError(f"Пароль довший за {BCRYPT_MAX_BYTES} байти (обмеження bcrypt)") 

54 return _password_hash.hash(password) 

55 

56 

57def verify_password(password: str, password_hash: str) -> bool: 

58 """False і для невірного пароля, і для хеша, якого не можна перевірити 

59 (напр. "!no-login" у користувача без пароля), і для пароля, довшого за 72 байти: 

60 такий пароль не міг бути збережений, а bcrypt на ньому падає (було б 500).""" 

61 if len(password.encode()) > BCRYPT_MAX_BYTES: 

62 return False 

63 try: 

64 return _password_hash.verify(password, password_hash) 

65 except UnknownHashError: 

66 return False 

67

69@lru_cache(maxsize=1) 

70def _dummy_hash() -> str: 

71 return _password_hash.hash("dummy-password-for-timing") 

72 

73 

74def verify_dummy_password(password: str) -> None: 

75 """Перевірка «навхолосту» для неіснуючого email: відповідь займає стільки ж часу, 

76 скільки для існуючого, тож за часом не можна дізнатися, чи зареєстрований email.""" 

77 verify_password(password, _dummy_hash())

80def create_access_token(user_id: str, role: str) -> str: 

81 now = datetime.now(timezone.utc) 

82 payload = { 

83 "sub": str(user_id), 

84 "role": role, 

85 "iss": JWT_ISSUER, 

86 "iat": now, 

87 "exp": now + access_token_ttl(), 

88 } 

89 return jwt.encode(payload, _secret(), algorithm=JWT_ALGORITHM)

92def decode_access_token(token: str) -> dict: 

93 """Перевіряє підпис, алгоритм, issuer і строк дії. Будь-яка помилка -> InvalidToken.""" 

94 try: 

95 return jwt.decode( 

96 token, 

97 _secret(), 

98 algorithms=[JWT_ALGORITHM], # явно: токени з alg=none чи іншим алгоритмом відхиляються 

99 issuer=JWT_ISSUER, 

100 options={"require": ["sub", "role", "iss", "iat", "exp"]}, 

101 ) 

102 except jwt.PyJWTError as exc: 

103 raise InvalidToken(str(exc)) from exc

106# Правила надійності пароля: (код порушення, перевірка, що правило виконано) 

107_PASSWORD_RULES = ( 

108 ("too_short", lambda p: len(p) >= 8), 

109 ("no_digit", lambda p: any(ch.isdigit() for ch in p)), 

110 ("no_letter", lambda p: any(ch.isalpha() for ch in p)), 

111 ("too_long", lambda p: len(p.encode()) <= BCRYPT_MAX_BYTES), 

112)

115def validate_password_strength(password: str) -> list[str]: 

116 """Повертає коди порушених правил; порожній список — пароль достатньо надійний.""" 

117 return [code for code, is_ok in _PASSWORD_RULES if not is_ok(password)]
